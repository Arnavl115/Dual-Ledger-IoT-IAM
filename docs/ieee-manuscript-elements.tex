% Consistent state and request notation.
\newcommand{\ledgerSet}{\mathcal{L}=\{\mathrm{F},\mathrm{I}\}}
\begin{align}
m &= \operatorname{UTF8}\!\left(
  \mathit{id}\parallel\texttt{:}\parallel a\parallel\texttt{:}\parallel t
\right), \label{eq:request-message}\\
\mathit{rid} &= \operatorname{SHA\text{-}256}(m), \label{eq:replay-id}\\
\operatorname{Fresh}(t) &\Longleftrightarrow
  T-W \le 1000t \le T+\delta, \label{eq:freshness}\\
\operatorname{Accept} &= V\land F\land L\land A\land S\land C, \label{eq:accept}\\
\mathcal{C}(d) &\Longleftrightarrow
  pk_{\mathrm{F}}(d)=pk_{\mathrm{I}}(d)
  \land s_{\mathrm{F}}(d)=s_{\mathrm{I}}(d), \label{eq:agreement}\\
T_{\mathrm{dual}} &= T_{\mathrm{F}}+T_{\mathrm{I}}+T_{\mathrm{P}}+T_{\mathrm{K}}.
\label{eq:dual-latency}
\end{align}
For each ledger $L\in\ledgerSet$, device state is
$D_L(d)=\bigl(pk_L(d),s_L(d)\bigr)$, where
$s_L(d)\in\{\textsc{active},\textsc{revoked}\}$.
